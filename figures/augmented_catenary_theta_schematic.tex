\begin{tikzpicture}

	\def\c{0.85}

	\def\xmincat{-1.5}
	\def\xmidcat{-1.0}
	\def\xmaxcat{1.0}

	\def\ymincat{0.0}
	\pgfmathsetmacro{\ymaxcat}{(cosh(\xmincat*\c)-1)/\c}
	\pgfmathsetmacro{\ymidcat}{(cosh(\xmaxcat*\c)-1)/\c}

	\def\buffer{.5}
	\def\xmin{\xmincat - \buffer}
	\def\xmax{\xmaxcat + \buffer}
	\def\ymin{\ymincat - \buffer}
	\def\ymax{\ymaxcat + \buffer}

	\def\arrowlength{0.25}

	\def\bgquadh{.1}
	\def\fgquadh{-\ymidcat}
	\pgfmathsetmacro{\abangle}{atan((\ymidcat - \ymaxcat) / (\xmaxcat - \xmincat))}
	\def\tiltangle{-2*\abangle}

	\pgfmathsetmacro{\tiltcatstretch}{(\xmaxcat - \xmincat) /((\xmaxcat - \xmidcat) * cos(\abangle))}

	\def\pointsize{2pt}

	\begin{axis}[
			xmin = \xmin,
			xmax = \xmax,
			ymin = \ymin,
			ymax = \ymax,
			width = 1.25\figurewidth,
			axis equal,
			clip = false,
			axis x line=none,
			axis y line=none
		]
		\coordinate (O) at (axis cs: 0.0, 0.0);
		\coordinate (x) at (axis cs: \arrowlength, 0.0);
		\coordinate (z) at (axis cs: 0.0, \arrowlength);

		\coordinate (A) at (axis cs: \xmincat, \ymaxcat);
		\coordinate (B) at (axis cs: \xmaxcat, \ymidcat);

		\coordinate (C) at ($(B) - (O) + (axis cs: 0.0, \bgquadh - \ymidcat + \ymaxcat)$);
		\coordinate (D) at ($(A) - (O) + (axis cs: 0.0, \bgquadh)$);

		\coordinate (E) at ($(A) - (O) + (axis cs: 0.0, \fgquadh + \ymidcat - \ymaxcat)$);
		\coordinate (F) at ($(B) - (O) + (axis cs: 0.0, \fgquadh)$);

		\coordinate (Ow) at (axis cs: {\xmin + \buffer / 2}, 0.0);
		\coordinate (xw) at (axis cs: {-\arrowlength * 0.9}, {-\arrowlength * 0.3});
		\coordinate (yw) at (axis cs: {\arrowlength * 0.9}, {-\arrowlength *  0.3});
		\coordinate (zw) at (axis cs: 0.0, \arrowlength );

		\coordinate (OTD) at (axis cs:
		{(\fgquadh + (\ymidcat - \ymaxcat) / 2) * sin(\tiltangle)},
		{(\fgquadh + (\ymidcat - \ymaxcat) / 2) * cos(\tiltangle)}
		);

		\coordinate (H) at (axis cs: 0, -\ymincat - \ymidcat);
		\coordinate (V) at (axis cs:{\xmincat + \xmaxcat - \xmidcat}, \ymaxcat);
		\coordinate (Ot) at ($(A)!.5!-\tiltangle:(V) - (O) + (axis cs: 0, 0)!1.!-\tiltangle:(H)$);
		\coordinate (xt) at ($(Ot) - (O) + (axis cs: {\arrowlength * cos(\tiltangle)}, {-\arrowlength * sin(\tiltangle) })$);
		\coordinate (zt) at ($(Ot) - (O) + (axis cs: {\arrowlength * sin(\tiltangle) }, {\arrowlength * cos(\tiltangle) })$);

		\fill[red!20, opacity=.9] (C) -- (D) -- (E) -- (F) -- cycle;

		\node[red, below left] at (C) {\lplane{\gamma}};

		\addplot[smooth, domain=\xmincat:\xmaxcat, samples=100, very
			thick, red] {{(cosh(\x*\c)-1)/\c}};

		\addplot[
		smooth, very thick, olive,
		domain=-\xmaxcat:-\xmincat, samples=100,
		shift={($(A) - (axis cs: {\xmidcat}, {(cosh(\xmidcat*\c)-1)/\c})$)},
		rotate around={2*\abangle:(A)}
		]
		{{(cosh(\x*\c)-1)/\c}};

		\filldraw (A) circle (\pointsize) node[above left] {\fpoint{A}{\gamma}};
		\filldraw (B) circle (\pointsize) node[above left] {\fpoint{B}{\gamma}};

		\coordinate (I) at (intersection of O--z and Ot--zt);
		\coordinate (I1) at ($(I) + 2*(O) - 2*(z)$);
		\coordinate (I2) at ($(I) + 2*(Ot) - 2*(zt)$);

		\draw[dotted] (I) -- (I1);
		\draw[dotted] (I) -- (I2);

		\filldraw (I) circle (\pointsize);
		\draw[thick, ->] (I) -- ($(I)!.5!(I1)$) node[right] {\ldvect{a}{\gamma}};
		\draw[thick, ->] (I) -- ($(I)!.5!(I2)$) node[left] {\ldvect{a}{\theta\gamma}};
		
		\pic [draw, ->, angle radius=1cm, angle eccentricity=1.2, "\footnotesize $\theta$"] {angle = I2--I--I1};
		
		\filldraw[red] (O) circle (\pointsize) node[above left] {\ldpoint{O}{\gamma}};
		\draw[red, thick, ->] (O) -- (x) node[above] {\ldvect{x}{\gamma}};
		\draw[red, thick, ->] (O) -- (z) node[above] {\ldvect{z}{\gamma}};

		\filldraw[olive] (Ot) circle (\pointsize) node[below left] {\ldpoint{O}{\theta\gamma}};
		\draw[olive, thick, ->] (Ot) -- (xt) node[below] {\ldvect{x}{\theta\gamma}};
		\draw[olive, thick, ->] (Ot) -- (zt) node[above] {\ldvect{z}{\theta\gamma}};

	\end{axis}
\end{tikzpicture}
