\begin{tikzpicture}

	\def\c{0.85}

	\def\xmincat{-1.5}
	\def\xmidcat{-1.0}
	\def\xmaxcat{1.0}

	\def\ymincat{0.0}
	\pgfmathsetmacro{\ymaxcat}{(cosh(\xmincat*\c)-1)/\c}
	\pgfmathsetmacro{\ymidcat}{(cosh(\xmaxcat*\c)-1)/\c}

	\def\buffer{.5}
	\def\xmin{\xmincat - \buffer}
	\def\xmax{\xmaxcat + \buffer}
	\def\ymin{\ymincat - \buffer}
	\def\ymax{\ymaxcat + \buffer}

	\def\arrowlength{0.25}

	\def\bgquadh{.1}
	\def\fgquadh{-\ymidcat}
	\def\tiltangle{20}
	\pgfmathsetmacro{\abangle}{atan((\ymidcat - \ymaxcat) / (\xmaxcat - \xmincat))}

	\pgfmathsetmacro{\tiltcatstretch}{(\xmaxcat - \xmincat) /((\xmaxcat - \xmidcat) * cos(\abangle))}

	\def\pointsize{2pt}

	\begin{axis}[
			xmin = \xmin,
			xmax = \xmax,
			ymin = \ymin,
			ymax = \ymax,
			width = 1.25\figurewidth,
			axis equal,
			clip = false,
			axis x line=none,
			axis y line=none
		]
		\coordinate (O) at (axis cs: 0.0, 0.0);
		\coordinate (x) at (axis cs: \arrowlength, 0.0);
		\coordinate (z) at (axis cs: 0.0, \arrowlength);

		\coordinate (A) at (axis cs: \xmincat, \ymaxcat);
		\coordinate (B) at (axis cs: \xmaxcat, \ymidcat);

		\coordinate (C) at ($(B) - (O) + (axis cs: 0.0, \bgquadh - \ymidcat + \ymaxcat)$);
		\coordinate (D) at ($(A) - (O) + (axis cs: 0.0, \bgquadh)$);

		\coordinate (E) at ($(B) - (O) + (axis cs: 0.0, \fgquadh)$);
		\coordinate (F) at ($(A) - (O) + (axis cs: 0.0, \fgquadh + \ymidcat - \ymaxcat)$);

		\coordinate (Ow) at (axis cs: {\xmin + \buffer / 2}, 0.0);
		\coordinate (xw) at (axis cs: {-\arrowlength * 0.9}, {-\arrowlength * 0.3});
		\coordinate (yw) at (axis cs: {\arrowlength * 0.9}, {-\arrowlength *  0.3});
		\coordinate (zw) at (axis cs: 0.0, \arrowlength );

		\coordinate (OTU) at (axis cs:
		{(\bgquadh - (\ymidcat - \ymaxcat) / 4) * sin(\tiltangle)},
		{(\bgquadh - (\ymidcat - \ymaxcat) / 4) * cos(\tiltangle)}
		);
		\coordinate (Ct) at ($(B) - (O) + (OTU)$);
		\coordinate (Dt) at ($(A) - (O) + (OTU)$);

		\coordinate (OTD) at (axis cs:
		{(\fgquadh + (\ymidcat - \ymaxcat) / 2) * sin(\tiltangle)},
		{(\fgquadh + (\ymidcat - \ymaxcat) / 2) * cos(\tiltangle)}
		);
		\coordinate (Et) at ($(B) - (O) + (OTD)$);
		\coordinate (Ft) at ($(A) - (O) + (OTD)$);

		\coordinate (Ot) at ($ (Et)!.5!(Ft) $);
		\coordinate (xt) at (axis cs: {\arrowlength * cos(-\abangle)}, {-\arrowlength * sin(-\abangle) });
		\coordinate (zt) at (axis cs: {\arrowlength * sin(\tiltangle) }, {\arrowlength * cos(\tiltangle) });

		\fill[red!20, opacity=.9] (A) -- (B) -- (Ct) -- (Dt) -- cycle;
		\fill[blue!20, opacity=.9] (A) -- (B) -- (C) -- (D) -- cycle;

		\node[blue, below left] at (C) {\lplane{v}};

		\addplot[smooth, domain=\xmincat:\xmaxcat, samples=100, very
			thick, blue] {{(cosh(\x*\c)-1)/\c}};

		\filldraw[blue] (O) circle (\pointsize) node[above left] {\ldpoint{O}{v}};
		\draw[blue, thick, ->] (O) -- (x) node[above] {\ldvect{x}{v}};
		\draw[blue, thick, ->] (O) -- (z) node[above] {\ldvect{z}{v}};

		\fill[blue!20, opacity=.5] (A) -- (B) -- (E) -- (F) -- cycle;
		\fill[red!20, opacity=.5] (A) -- (B) -- (Et) -- (Ft) -- cycle;

		\node[red, above left] at (Et) {\lplane{\gamma}};

		\addplot[
		smooth, very thick, red,
		domain=\xmidcat*\tiltcatstretch:\xmaxcat*\tiltcatstretch, samples=100,
		shift={($(A) - (axis cs: {\xmidcat*\tiltcatstretch}, {(cosh(\xmidcat*\tiltcatstretch*\c)-1)/\c})$)},
		rotate around={\abangle:(A)}
		]
		{{(cosh(\x*\c)-1)/\c}};

		
		\filldraw[red] (Ot) circle (\pointsize) node[below left] {\ldpoint{O}{\gamma}};
		\draw[red, thick, ->] (Ot) -- ($(Ot) - (O) + (xt)$) node[below] {\ldvect{x}{\gamma}};
		\draw[red, thick, ->] (Ot) -- ($(Ot) - (O) + (zt)$) node[above] {\ldvect{y}{\gamma}};
		
		\coordinate (Ad) at ($(A) + 1.5*(O) - 1.5*(z)$);
		\coordinate (At) at ($(A) + 1.5*(O) - 1.5*(zt)$);

		\draw[dotted] (A) -- (Ad);
		\draw[dotted] (A) -- (At);

		\draw[thick, ->, bend left=\tiltangle] (Ad) to node[midway, below] {$\gamma$} (At);

		\draw[thick, ->] (A) -- ($(A) - (O) + (xt)$) node[above] {\vect{e}};
		\draw[thick, ->] (A) -- ($(A) + (O) - (z)$) node[below right] {\vect{g}};
		\draw[thick, ->] (A) -- ($(A) + (O) - (zt)$) node[left] {\ldvect{a}{\gamma}};

		\filldraw (A) circle (\pointsize) node[above left] {\fpoint{A}{v}};
		\filldraw (B) circle (\pointsize) node[above left] {\fpoint{B}{v}};

		\filldraw (Ow) circle (\pointsize) node[below] {\ldpoint{O}{w}};
		\draw[thick, ->] (Ow) -- ($(Ow) - (O) + (xw)$) node[above] {\ldvect{x}{w}};
		\draw[thick, ->] (Ow) -- ($(Ow) - (O) + (yw)$) node[above] {\ldvect{y}{w}};
		\draw[thick, ->] (Ow) -- ($(Ow) - (O) + (zw)$) node[above] {\ldvect{z}{w}};

	\end{axis}
\end{tikzpicture}
