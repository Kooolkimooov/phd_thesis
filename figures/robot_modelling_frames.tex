\begin{tikzpicture}

	\def\arrowlength{.2}

	\begin{axis}[
			width=\figurewidth,
			axis equal,
			clip=false,
			axis x line=none,
			axis y line=none,
			xmin=-1.0, xmax=1.0,
			ymin=-1.0, ymax=1.0,
			trig format=rad
		]

		\coordinate (Oi) at (axis cs: 0.0, 0.0);
		\coordinate (xi) at (axis cs: \arrowlength, 0.0);
		\coordinate (zi) at (axis cs: 0.0, \arrowlength);
		\coordinate (yi) at ($(xi)!0.5!(zi)$);
		% schematic water surface at y=0 (small amplitude, fast/slow components)
		\addplot[domain=-1.0:1.0, samples=300, draw=blue!40, very thick] ({\x},{0.045*sin(6*\x)});
		\addplot[domain=-1.0:1.0, samples=300, draw=blue!20, thick] ({\x},{0.02*sin(14*\x)});

		\draw[->, thick, blue] (Oi) -- (xi) node[below right] {\ldvect{x}{W}};
		\draw[->, thick, blue] (Oi) -- (yi) node[right] {\ldvect{y}{W}};
		\draw[->, thick, blue] (Oi) -- (zi) node[above right] {\ldvect{z}{W}};
		\filldraw[blue] (Oi) circle (2pt) node[below left] {\lframe{W}};

		\coordinate (Ob) at (axis cs: 0.5, -0.5);
		\coordinate (xb) at (axis cs: -\arrowlength, 0.0);
		\coordinate (zb) at (axis cs: 0.0, \arrowlength);
		\coordinate (yb) at ($(xb)!0.5!(zb)$);

		\draw[->, thick, red] (Ob) -- ($(Ob) - (Oi) + (xb)$) node[below left] {\ldvect{x}{B}};
		\draw[->, thick, red] (Ob) -- ($(Ob) - (Oi) + (yb)$) node[left] {\ldvect{y}{B}};
		\draw[->, thick, red] (Ob) -- ($(Ob) - (Oi) + (zb)$) node[above left] {\ldvect{z}{B}};
		\filldraw[red] (Ob) circle (2pt) node[below right] {\lframe{B}};
	\end{axis}

\end{tikzpicture}
